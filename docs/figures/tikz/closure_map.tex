% closure_map.tex -- the Track-2 (Paper #2) identifiability-limits map: the three BGC
% closures real observations could target, and the distinct reason each cannot be sharply
% constrained. Faithful to the write-up: iron = observability wall (structural), calcite
% Omega-modulation = data/support-limited (within-region Omega <= 0.16 dex; out-of-sample
% E2 negative), growth = structurally unobservable (NPP gives only the biomass-weighted mean).
% Equal-height, top-aligned cards (fixed-height minipages) so nothing overlaps the verdict band.
% Standalone: `pdflatex closure_map.tex`.
\documentclass[border=8pt]{standalone}
\usepackage{darwindiff-tikz}
\begin{document}
\begin{tikzpicture}
  \tikzset{closurecard/.style={draw=ddLine, line width=0.6pt, rounded corners=3pt,
    fill=white, inner sep=8pt, anchor=north west}}
  \newcommand{\cardh}{4.75cm}
  \newcommand{\cardw}{4.15cm}

  \node[ddlabel, text=ddMuted, anchor=south west] (title) at (0,3.2)
    {Track 2 --- what real observations can (and cannot) constrain in Darwin's BGC closures};

  % ---- iron: observability wall -------------------------------------------------
  \node[closurecard] (iron) at (0,2.7) {\begin{minipage}[t][\cardh][t]{\cardw}
    {\sffamily\small\textbf{iron}}\hfill{\ttfamily\footnotesize scav\_rat}\\[3pt]
    {\sffamily\scriptsize\bfseries\color{ddLimit} observability-limited (structural)}\\[6pt]
    {\sffamily\footnotesize observable: DFe concentration}\\[5pt]
    {\sffamily\scriptsize\color{ddMuted} concentration under-determines the scavenging \emph{rate} --- an information wall (Tagliabue 2016).}\\[5pt]
    {\sffamily\scriptsize\color{ddMuted} \emph{evidence:} flat profile 25/32/100$\times$; a particulate:dissolved observable does not rescue it.}
  \end{minipage}};

  % ---- calcite: data / support-limited ------------------------------------------
  \node[closurecard] (calc) at ([xshift=5mm]iron.north east) {\begin{minipage}[t][\cardh][t]{\cardw}
    {\sffamily\small\textbf{calcite}}\hfill{\ttfamily\footnotesize R\_PICPOC}\\[1pt]
    {\sffamily\scriptsize\color{ddMuted}\hfill $\Omega$-modulation}\\[3pt]
    {\sffamily\scriptsize\bfseries\color{ddAccent} data / support-limited}\\[6pt]
    {\sffamily\footnotesize observable: PIC:POC ratio}\\[5pt]
    {\sffamily\scriptsize\color{ddMuted} within-region $\Omega$ support $\le 0.16$ dex everywhere --- too narrow to fit $\mathrm{ratio}=R_0\,\Omega^{n}$.}\\[5pt]
    {\sffamily\scriptsize\color{ddMuted} \emph{evidence:} independent GLODAPv3 $\Omega$ + driver test + out-of-sample E2 (negative).}
  \end{minipage}};

  % ---- growth: structurally unobservable ----------------------------------------
  \node[closurecard] (grow) at ([xshift=5mm]calc.north east) {\begin{minipage}[t][\cardh][t]{\cardw}
    {\sffamily\small\textbf{growth}}\hfill{\ttfamily\footnotesize Smallgrow, Biggrow}\\[3pt]
    {\sffamily\scriptsize\bfseries\color{ddUnobs} structurally unobservable}\\[6pt]
    {\sffamily\footnotesize observable: NPP / biomass}\\[5pt]
    {\sffamily\scriptsize\color{ddMuted} total NPP constrains only the biomass-weighted \emph{mean} --- the pair stays degenerate.}\\[5pt]
    {\sffamily\scriptsize\color{ddMuted} \emph{evidence:} loss-degenerate; needs a per-PFT production observable (none exists).}
  \end{minipage}};

  % ---- unifying verdict band (below all three equal-height cards) ----------------
  \node[verdict=ddAccent, fill=ddAccentBg, anchor=north west, inner sep=8pt]
        (band) at ([yshift=-6mm]iron.south west) {%
    \begin{minipage}{13.9cm}
    \sffamily\footnotesize
    Across all three, real observations do \textbf{not} sharply constrain the closure --- for \emph{three distinct reasons}. The binding constraint is the \textbf{observing system, not the method}.\\[3pt]
    \textbf{Forward contribution:} an observing-system recommendation --- wider within-region $\Omega$ $\cdot$ pure scavenged Fe $\cdot$ per-PFT production.
    \end{minipage}};

\end{tikzpicture}
\end{document}
