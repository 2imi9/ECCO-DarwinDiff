% fig2_identifiability_v2_basinmatrix.tex -- IDENTIFIABILITY MAP, variant 2: "basin matrix".
%
% Why this variant exists. The banded table (variant 1) states the headline per-AOI counts but
% has to relegate the BASIN STRUCTURE to prose -- and the basin structure is the actual finding:
% recovery is not uniform, and the >=2-of-3 metric's output is a direct consequence of which
% basins each parameter lands in. This figure shows that mechanism instead of asserting it.
%
% Numbers: single-AOI Cal+ counts (within 40% of Carroll), n=50, re-derived from the per-seed
% `per_aoi_recovered` records of the flagship run n50e2k_percell_trio (2000 epochs).
%
% Greyscale safety: the cell value is printed IN the cell, and the fill is a single-hue
% sequential ramp (opacity by value), so the figure carries no information in hue alone.
\documentclass[border=6pt]{standalone}
\usepackage{darwindiff-tikz}
\usepackage{amsmath}

\begin{document}

                   minimum height=1.10cm, anchor=center},
  cnum/.style   = {ddfont, font=\sffamily\small\bfseries, text=ddInk},
  rowlab/.style = {font=\ttfamily\small, text=ddInk, anchor=east},
  collab/.style = {ddfont, font=\sffamily\footnotesize\bfseries, text=ddMuted,
                   align=center, text width=2.3cm},
]

       best-conditioned iron basin) and barely at all in the equatorial Pacific (7/50), which
       stays ratio-degenerate even at 4000 epochs. \texttt{diatomgraz} inverts that pattern,
