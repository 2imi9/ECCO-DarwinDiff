% readme_components.tex -- README Figure 2, the "differentiable table".
%
% What it shows: every component of the flagship DarwinDiff training step, one row each, in an
% ML-paper (booktabs) method table. The columns say whether a component is LEARNED, whether it is
% DIFFERENTIABLE (sits on the gradient path), whether it varies PER CELL or is shared, and where
% it comes from. Exactly one row is learned (the per-cell 1x1-conv network, set in bold on an
% opaque ddWash band: the "ours highlighted" idiom); theta is its bounded output. The model group
% runs in data-flow order: network -> bounds map -> theta -> box step, then the fixed constants.
%
% Canvas: 499 x 249 bp (176.04 x 87.84 mm) instead of 176 x 88 mm. This pdftocairo build fits a
% page whose size is not a whole number of bp onto a whole-bp page, shrinking the SVG by 0.22%
% and leaving a ~0.5 pt unpainted sliver at its left, right and bottom edges; a whole-bp page
% gives the SVG an identity transform, so the committed SVG matches the PDF.
%
% Rules it follows (dd-readme.sty):
%   * red = the gradient: here ONLY the thin check strokes; never a fill or a text colour.
%   * blue bar = a real observation; gold bar = ECCO-Darwin v05 model output. Full strength.
%   * "Differentiable" is defined on components and loss TERMS, never on data.
%   * booktabs look: no vertical rules, no zebra, no row outlines, no shadows, no opacity; the
%     only shading is the one learned-row band, drawn before the rules. Top, mid and bottom rules
%     span x 1.5-174.5 so the rotated group tags sit inside the frame (a multirow group column);
%     the group separators are partial, from x 7. Every cell sits on the column grid.
%   * no result numbers: the only numerals are "1x1", the citation years and the product name
%     "ECCO-Darwin v05". No counts, weights, tracer or step numbers.
%   * the Per cell column is one glyph family, a 3x3 hairline grid (ddMute stroke): light grey
%     texture = varies by cell, one uniform grey = one shared value, two ink cells = sparse
%     observations (all three keyed in the legend).
%   * 8 pt floor, drawn at final size (176 mm = the README column), opaque white canvas.
%
% Deviations from the plan spec, each from the round-1 review:
%   - no type-glyph column (the v2 shape-grammar squares): at README width it read as a column of
%     unchecked checkboxes, it had no key, and the Learned dot/ring and the gold/blue bars already
%     carry learned vs data vs source. Names move from x 16 to x 12 (2.9 mm right of the bars).
%   - no dashed outlines on the two scope rows: with the glyphs they drew a doubled dashed wall
%     and stacked six horizontal strokes in the bottom 10 mm. The scope group is marked by its
%     tag, its separator and italic ddMute names, like any other booktabs group.
%   - the lateral-transport row sits on the column grid (dashes in Learned, Differentiable and
%     Per cell, a note in Source) instead of one spanning line; the note says what is absent,
%     "no exchange between cells", since cells ARE coupled through the shared weights and the
%     per-region z-scoring of the pattern term.
%   - glyph columns re-spaced to even header gutters (72 / 95 / 118, Source at 132), so the
%     table's mass centres on the canvas instead of hugging the left.
%   - theta is Euler (eulergreek option), as in readme_method; plain \sansmath leaves it serif.
%   - theta's source is "network + bounds map": theta here, as in readme_method, is the bounded
%     parameter field that enters the box.
%   - the learned dot is 2.6 mm (spec 2.2) so the one learned mark is the darkest in its column.
%   - the GRID greys are white/C8/7F (spec ED/D9/C4) so "varies by cell" reads as a checkered
%     texture at README width, clearly apart from the flat grey of SHARED; SHARED is the same grid filled uniformly (spec: a 1.6 mm ddMute square).
%   - the legend drops the learned dot and the red check (their column headers say it) and is
%     centred on the rule span instead of starting at 16.
%   - the learned row's source is "trained here" (spec "this repo") and its sub-label says the
%     weights are shared; both scope rows set their Source in italic ddMute, like their names.

\definecolor{ddV2}{HTML}{C8C8C8}
\definecolor{ddV3}{HTML}{7F7F7F}

\tikzset{
  % one text style per role; "mid" anchors keep baselines level whatever the descenders
  cHdr/.style   = {font=\fLabel\bfseries, text=ddInk, inner sep=0pt},
  cName/.style  = {font=\fLabel, text=ddInk, inner sep=0pt, anchor=mid west},
  cSub/.style   = {font=\fSmall, text=ddMute, inner sep=0pt, anchor=base west},
  cSrc/.style   = {font=\fSmall, text=ddInk, inner sep=0pt, anchor=mid west},
  cTag/.style   = {font=\fSmall, text=ddMute, inner sep=0pt, rotate=90},
  % ---- Learned column ----
  pics/dot/.style  = {code={\fill[ddInk] (0,0) circle (1.3);}},
  pics/ring/.style = {code={\draw[ddInk, line width=0.6pt] (0,0) circle (1.1);}},
  pics/no/.style   = {code={\draw[ddRule, line width=0.6pt] (-1.1,0) -- (1.1,0);}},
  % ---- Differentiable column: the only red in the figure ----
  pics/tick/.style = {code={\draw[ddRed, line width=0.9pt, line cap=round, line join=round]
                            (-1.3,0.1) -- (-0.3,-1.0) -- (1.5,1.3);}},
  % ---- Per cell column ----
  % One glyph family: the same 3x3 hairline grid of 1.1 mm cells, ddMute 0.3pt stroke; only
  % the fill says varies / shared / sparse.
  % GRID: 3x3 opaque greys, a Latin square, so "varies by cell" reads without colour
  pics/grid/.style = {code={
      \foreach \c [count=\i from 0] in {white,ddV2,ddV3,ddV2,ddV3,white,ddV3,white,ddV2}{
        \pgfmathtruncatemacro{\col}{mod(\i,3)}\pgfmathtruncatemacro{\row}{div(\i,3)}
        \fill[\c] (-1.65+1.1*\col, 1.65-1.1*\row) rectangle ++(1.1,-1.1);}
      \pic {cellgrid};}},
  % SHARED: one value for every cell, all nine cells the same grey (the GRID's darkest, so the
  % flat glyph stays distinct from the light texture at README width)
  pics/shared/.style = {code={
      \fill[ddG3] (-1.65,-1.65) rectangle (1.65,1.65);
      \pic {cellgrid};}},
  % SPARSE: observations exist at a few cells only
  pics/sparse/.style = {code={
      \fill[ddInk] (-1.65,1.65) rectangle ++(1.1,-1.1);   % top-left
      \fill[ddInk] (0.55,0.55) rectangle ++(1.1,-1.1);    % middle-right
      \pic {cellgrid};}},
  pics/cellgrid/.style = {code={
      \draw[ddMute, line width=0.3pt, step=1.1, shift={(-1.65,-1.65)}] (0,0) grid (3.3,3.3);}},
  % "mid" like every other cell, so its baseline sits level with the row's names and sources
  pics/perregion/.style = {code={
      \node[font=\fSmall, text=ddMute, inner sep=0pt, anchor=mid] {per region};}},
  % ---- colour bar (source category), 2.1 x 4.0 mm, full strength ----
  pics/bar-gold/.style = {code={\fill[ddGold] (0,-2.0) rectangle (2.1,2.0);}},
  pics/bar-blue/.style = {code={\fill[ddBlue] (0,-2.0) rectangle (2.1,2.0);}},
  pics/bar-none/.style = {code={}},
  % legend keys for the bars: 2.1 x 3.0 mm, centred
  pics/key-gold/.style = {code={\fill[ddGold] (-1.05,-1.5) rectangle (1.05,1.5);}},
  pics/key-blue/.style = {code={\fill[ddBlue] (-1.05,-1.5) rectangle (1.05,1.5);}},
}

% \ddrow{y}{bar}{name}{sub-label}{learned}{differentiable}{per cell}{source}
%   bar in {gold, blue, none}; the three glyph columns take pic names. The name and source
%   arguments may open with a font or colour switch (\bfseries, \itshape\color{ddMute}).
\newcommand{\ddrow}[8]{%
  \pic at (\xBar,#1)  {bar-#2};
  \node[cName] (nm) at (\xName,#1) {#3};
  \node[cSub] at ($(nm.base east)+(1.5,0)$) {#4};
  \pic at (\xLrn,#1) {#5};
  \pic at (\xDif,#1) {#6};
  \pic at (\xCel,#1) {#7};
  \node[cSrc] at (\xSrc,#1) {#8};}
\newcommand{\scopetext}{\itshape\color{ddMute}}

% \legitem{previous node}{glyph width}{pic}{text}{this node}: glyph, 1 mm gap, text; items 3 mm apart
\newcommand{\legitem}[5]{%
  \path let \p1 = (#1.east) in coordinate (lg) at (\x1+3mm+#2mm/2, 4.6mm);
  \pic at (lg) {#3};
  \node[font=\fSmall, text=ddInk, inner sep=0pt, anchor=mid west] (#5)
    at ($(lg)+(#2/2+1,0)$) {#4};}
